  \end{columns}

\begin{frame}{The Most Likely Outcome Does Not Settle the Choice}
  The shortcut is more likely to be clear than blocked. But we also need to know:
  \begin{itemize}
    \item How much time does a successful shortcut save?
    \item What happens if the attempt fails?
    \item How valuable is a predictable arrival?
  \end{itemize}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{block}{Probability}
        How likely is each consequence?
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Preference}
        How desirable is each consequence?
      \end{block}
    \end{column}
  \end{columns}
  \begin{keyidea}
    An agent needs both a model of what may happen and a model of what matters.
  \end{keyidea}
\end{frame}

\begin{frame}{Keep One Decision in View}
  For this lecture, compare two routes for \textbf{one delivery segment}.
  \begin{itemize}
    \item Choosing a route does not change whether the shortcut is currently blocked.
    \item Each outcome score includes the segment's time, energy, and disruption.
    \item The detour's outcome is fixed in this simplified model.
    \item We will also ask whether information is worth buying \emph{before} the route choice.
  \end{itemize}
  \begin{keyidea}
    We finish evaluating the example at the end of this segment. Choosing what to do at every later intersection belongs to Lecture 23.
  \end{keyidea}
\end{frame}

\section{Preferences Become Utility}

\begin{frame}{Utility Makes the Tradeoff Explicit}
  A \textbf{utility} is a score for how desirable an outcome is to the agent.
  \begin{center}
    \renewcommand{\arraystretch}{1.5}
    \begin{tabular}{l|r|l}
      \textbf{Outcome} & \textbf{Utility} & \textbf{Interpretation} \\\hline
      Shortcut succeeds & $80$ & fast, uneventful arrival \\
      Take the detour & $45$ & slower, predictable arrival \\
      Shortcut is blocked & $-60$ & substantial delay and disruption
    \end{tabular}
  \end{center}
  These are illustrative preference scores, not probabilities or minutes.
  \begin{keyidea}
    Higher means preferred. The gaps also matter: they express how much upside compensates for the downside in uncertain choices.
  \end{keyidea}
\end{frame}

\begin{frame}{The Scores Are a Modeling Choice}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{block}{Flexible delivery}
        A failed shortcut is inconvenient. Saving time may justify more risk.
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Time-critical delivery}
        A failed attempt could miss the delivery window. Reliability may matter more.
      \end{block}
    \end{column}
  \end{columns}
  The blockage probabilities can be identical while the preferred route differs.
  \begin{activity}
    Which outcome score would you reconsider if lateness became much more costly? Why would that change the decision?
  \end{activity}
  \begin{keyidea}
    An accurate probability model cannot decide which outcomes the system should value.
  \end{keyidea}
\end{frame}

\begin{frame}{Expected Utility Is a Probability-Weighted Score}
  Compare the possible consequences of each action using the same utility scale.
  \[
    \text{expected utility}
    =\sum_{\text{possible outcomes}}
      \text{probability}\times\text{utility}.
  \]
  In words:
  \begin{enumerate}
    \item List the consequences an action could produce.
    \item Weight each consequence's utility by its probability.
    \item Add the contributions and compare feasible actions.
  \end{enumerate}
  \begin{keyidea}
    This is the weighted-average idea from expectimax, using beliefs from our probability model.
  \end{keyidea}
\end{frame}

\begin{frame}{Worked Choice: Shortcut or Detour?}
  \begin{center}
    \renewcommand{\arraystretch}{1.5}
    \begin{tabular}{l|rr}
      \textbf{Action} & \textbf{Clear: 65\%} & \textbf{Blocked: 35\%} \\\hline
      Shortcut & $80$ & $-60$ \\
      Detour & $45$ & $45$
    \end{tabular}
  \end{center}
  \[
    EU(\text{Shortcut})=0.65(80)+0.35(-60)=31
  \]
  \[
    EU(\text{Detour})=45
  \]
  Here $EU$ means expected utility. Choose the detour: $45>31$.
  \begin{keyidea}
    The shortcut's likely success does not outweigh its costly failure under these preferences.
  \end{keyidea}
\end{frame}

\begin{frame}{The Same Calculation as an Outcome Tree}
  \begin{center}
    \begin{tikzpicture}[x=1cm,y=1cm]
      \node[decisionchoice] (choice) at (0,0) {Choose};
      \node[decisionchance] (shortcut) at (3,0.8) {Shortcut};
      \node[decisionutility] (clear) at (7,1.6) {$80$};
      \node[decisionutility] (blocked) at (7,0) {$-60$};
      \node[decisionutility] (detour) at (3,-1.4) {$45$};

    \end{tikzpicture}
  \end{center}
  \begin{keyidea}
    At chance branches, take a weighted average. At a choice, select the action with the higher expected utility.
  \end{keyidea}
\end{frame}

\begin{frame}{Better Evidence Can Change the Preferred Action}
  Keep the same utilities and vary only the chance that the shortcut is clear.
  \begin{center}
    \renewcommand{\arraystretch}{1.5}
    \begin{tabular}{c|r|r|l}
      \textbf{Chance clear} & \textbf{Shortcut EU} & \textbf{Detour EU} & \textbf{Choice} \\\hline
      65\% & $31$ & $45$ & Detour \\
      75\% & $45$ & $45$ & Tie \\
      85\% & $59$ & $45$ & Shortcut
    \end{tabular}
  \end{center}
  \begin{activity}
    Why does ``more likely clear than blocked'' provide too little information to justify the shortcut?
  \end{activity}
\end{frame}

\begin{frame}{A Good Decision Can Still Miss the Best Outcome}
  Suppose the robot takes the detour and later learns the shortcut was clear.
  \begin{itemize}
    \item That does not make the original decision irrational.
    \item At decision time, the robot had to account for the blocked outcome too.
    \item We should assess the choice using the information available when it was made.
  \end{itemize}
  \begin{keyidea}
    Expected utility compares choices before uncertainty is resolved. It does not promise the best result on every individual attempt.
  \end{keyidea}
\end{frame}

\begin{frame}{What Should Go Into the Utility Model?}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item \textbf{Time:} travel time, deadlines, and tolerance for delay.
        \item \textbf{Energy:} battery cost, slopes, and the reserve needed to finish.
        \item \textbf{Reliability:} failed attempts, disruption, and predictable completion.
      \end{itemize}
      A weighted score is one option. Its attributes and scales must reflect the intended tradeoffs.
    \end{column}
    \begin{column}{0.47\textwidth}
      \lectureimage[2.05in]{utility-tradeoffs.png}
    \end{column}
  \end{columns}
  \begin{keyidea}
    Utility design turns an informal goal such as ``make a good delivery'' into explicit preferences we can inspect and challenge.
  \end{keyidea}
\end{frame}

\begin{frame}{Some Requirements Define Which Actions Are Allowed}
  Before comparing utilities, identify the \textbf{feasible actions}.
  \begin{itemize}
    \item A closed walkway can rule out a route entirely.
    \item Insufficient battery can rule out a long detour.
    \item A safety envelope can rule out a maneuver near pedestrians.
  \end{itemize}
  Then compare the expected utility of the remaining choices.
  \begin{activity}
    Should a prohibited route become acceptable merely because it saves enough time? What should the action set represent?
  \end{activity}
  \begin{keyidea}
    Some requirements belong in constraints; other preferences allow tradeoffs within those constraints.
  \end{keyidea}
\end{frame}

  \begin{center}

    \end{tikzpicture}
  \end{center}
  \begin{keyidea}
    A decision network, also called an influence diagram, adds choices and preferences to a probabilistic model.
  \end{keyidea}
\end{frame}

\begin{frame}{Our Route Choice as a Decision Network}
  \begin{center}
    \begin{tikzpicture}
      \node[decisionchance,minimum width=2.5cm] (b) at (0,1) {Blocked?};
      \node[decisionchoice,minimum width=2.5cm] (a) at (0,-1) {Route};
      \node[decisionutility,minimum width=3cm] (u) at (5,0) {Utility};
      \draw[decisionlink] (b) -- (u);
      \draw[decisionlink] (a) -- (u);
    \end{tikzpicture}
  \end{center}
  \begin{itemize}
    \item The score depends on the route and the true blockage state.
    \item The robot chooses the route; it does not choose the blockage state.
    \item Blockage is hidden, so the robot averages over its belief.
  \end{itemize}
  \begin{keyidea}
    Arrows into utility show what determines the score. The diagram does not make the hidden state observable.
  \end{keyidea}
\end{frame}

\begin{frame}{Add a Reading Available Before the Choice}
  \begin{center}
    \begin{tikzpicture}

  \end{center}
  \begin{itemize}
    \item The sensor reading depends probabilistically on blockage.
    \item The dashed information arrow says the reading is available to the decision.
    \item An imperfect reading still leaves uncertainty about blockage.
  \end{itemize}
\end{frame}

\begin{frame}{Use Information the Agent Actually Has}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{block}{Before a reading}
        Compare routes using the current 65\% / 35\% belief.
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{After a reading}
        Update the belief, then compare routes again. Different readings can justify different choices.
      \end{block}
    \end{column}
  \end{columns}
  \begin{activity}
    Why would ``choose the shortcut when it is clear and the detour when it is blocked'' give the robot an unfair advantage if it has no sensor reading?
  \end{activity}
  \begin{keyidea}
    Average over what remains hidden. A choice may depend only on information available before it is made.
  \end{keyidea}
\end{frame}

\section{Why Information Can Be Worth Buying}

\begin{frame}{A Scan Is Useful When It Improves the Route Choice}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.48\textwidth}
      A scan costs time and energy. Its benefit is a better-informed decision.
      \begin{enumerate}
        \item Acquire a reading.
        \item Update the belief.
        \item Choose a route using that information.
      \end{enumerate}
      Compare this plan with acting now, using the same outcome utilities and including the scan cost.
    \end{column}
    \begin{column}{0.47\textwidth}
      \lectureimage[2.05in]{scan-before-choice.png}
    \end{column}
  \end{columns}
  \begin{keyidea}
    Value of information asks how much better the decision could become, rather than how much data the sensor produces.
  \end{keyidea}
\end{frame}

\begin{frame}{Start with an Ideal Scan That Reveals Blockage Perfectly}
  \begin{center}
    \begin{tikzpicture}
      \node[decisionchance] (scan) at (0,0) {Perfect scan};
      \node[decisionchoice] (clear) at (4,1) {Choose shortcut};
      \node[decisionchoice] (blocked) at (4,-1) {Choose detour};
      \node[decisionutility] (good) at (8,1) {$80$};
      \node[decisionutility] (safe) at (8,-1) {$45$};
      \draw[decisionlink] (scan) -- node[above,sloped,font=\small] {clear: 65\%} (clear);
      \draw[decisionlink] (scan) -- node[below,sloped,font=\small] {blocked: 35\%} (blocked);
      \draw[decisionlink] (clear) -- (good);
      \draw[decisionlink] (blocked) -- (safe);
    \end{tikzpicture}
  \end{center}
  Assume the scan leaves blockage and route availability unchanged; its cost captures the delay and energy.
  \begin{keyidea}
    We learn first, then choose. Perfect information lets the robot avoid the blocked-shortcut outcome entirely.
  \end{keyidea}
\end{frame}

\begin{frame}{How Much Would Perfect Information Be Worth?}
  Without scanning, the best route has expected utility $45$.
  \[
    \text{With perfect information, before cost}
      =0.65(80)+0.35(45)=67.75.
  \]
  \[
    \text{Gross value of information}=67.75-45=22.75.
  \]
  \begin{itemize}
    \item If the scan costs $5$ utility units: $67.75-5=62.75$; scan.
    \item If it costs $25$: $67.75-25=42.75$; take the detour now.
  \end{itemize}
  \begin{keyidea}
    Perfect information about blockage is worth at most $22.75$ here. A noisy scan of that same variable cannot have greater gross value under the same assumptions.
  \end{keyidea}
\end{frame}

\begin{frame}{More Accurate Information Can Still Be Useless for This Choice}
  Now consider a weak sensor whose readings never make the shortcut preferable to the detour.
  \begin{itemize}
    \item The scan may improve its estimate of blockage.
    \item The route choice still stays the same.
    \item Before acquisition cost, the expected improvement in decision utility is zero.
  \end{itemize}
  \begin{activity}
    Can you name a measurement that helps describe the world but would not affect this particular route decision?
  \end{activity}
  \begin{keyidea}
    Under these assumptions, free optional information cannot hurt because it can be ignored. Paying to obtain it can still be a poor choice.
  \end{keyidea}
\end{frame}

\begin{frame}{What Changes with a Real, Noisy Sensor?}
  The reasoning stays the same:
  \begin{enumerate}
    \item Consider each reading the sensor might produce.
    \item Update the belief for that reading.
    \item Choose the best route under the updated belief.
    \item Average across readings, then subtract the scan cost.
  \end{enumerate}
  A noisy reading may leave enough uncertainty that the detour remains best.
  \begin{keyidea}
    The important idea is the conditional choice after a reading. The detailed Bayes calculations are in the optional backup slides.
  \end{keyidea}
\end{frame}

\begin{frame}{Discussion: Which Scan Would You Buy?}
  The route, prior belief, and utilities stay as in our example.
  \begin{center}
    \renewcommand{\arraystretch}{1.5}
    \begin{tabular}{l|l|p{0.42\textwidth}}

\begin{frame}{The Boundary: One Route Choice Versus a Continuing Strategy}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{block}{Lecture 22}
        We score the consequences of a specified route choice, possibly after acquiring information.

        We stop at the end of the segment.
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Lecture 23}

      \end{block}
    \end{column}
  \end{columns}
  \begin{keyidea}
    The distinction is the planning problem we model. Decision networks can also represent multiple decisions; our examples today deliberately isolate one route choice.
  \end{keyidea}
\end{frame}

\begin{frame}{Next: Actions Change What Becomes Possible Later}
  \begin{columns}[c,onlytextwidth]

        \item Charging costs time now but may make later destinations reachable.

\begin{frame}{Takeaways: Beliefs, Preferences, and a Choice}
  \begin{itemize}
    \item Probability tells us what may happen; utility expresses what matters.
    \item Expected utility weights each consequence's utility by its probability.
    \item Decision networks make uncertainty, controllable choices, and utility explicit.
    \item Information is valuable through the decisions it can improve, after accounting for its cost.
    \item A good decision uses the information available when it is made.
  \end{itemize}
  \begin{keyidea}
    Today: compare consequences for a specified choice. Next: plan how to act over time, with MDPs and POMDPs.
  \end{keyidea}
\end{frame}

\begin{frame}[t]{Backup: Under Uncertainty, Use Expected Utility}
  \small
  Average the utility of each possible outcome, weighted by its probability:
  \[
    EU(a\mid e)=\sum_o P(o\mid a,e)U(o).
  \]
  Then choose among feasible actions:
  \[
    a^*=\arg\max_{a\in\mathcal{A}} EU(a\mid e).
  \]
  \equationlegend{$a$: candidate action; $e$: evidence already observed; $o$: possible outcome; $P(o\mid a,e)$: outcome probability under the chosen action and evidence. $\sum_o$: sum over outcomes; $\mathcal{A}$: feasible action set; $\arg\max$: return a maximizing action; $a^*$: chosen action.}
  \begin{keyidea}
    Probability supplies the weights; utility supplies the values. We will suppress fixed evidence $e$ in the numerical example.
  \end{keyidea}
\end{frame}

\begin{frame}[t]{Backup: When Would We Switch to the Shortcut?}
  \small
  Let $p$ be the probability that the shortcut is clear. It beats the detour when
  \[
    \underbrace{80p-60(1-p)}_{EU(\text{Shortcut})}>\underbrace{45}_{EU(\text{Detour})}.
  \]
  Rearranging gives
  \[
    140p>105\qquad\Longrightarrow\qquad p>0.75.
  \]
  At $p=0.75$ the actions tie; at our current $p=0.65$, choose the detour.
  \equationlegend{$p=P(\text{Clear})$; $1-p=P(\text{Blocked})$; $80,-60,45$: utilities from the table. $EU$: expected utility; $>$: strict preference here; $\Longrightarrow$: implies.}
  \begin{keyidea}
    ``More likely clear than blocked'' is insufficient. These utilities require more than a 75\% chance of a clear shortcut.
  \end{keyidea}
\end{frame}

\begin{frame}[t]{Backup: Changing the Utility Scale Preserves the Decision}
  \small
  Under the standard expected-utility consistency assumptions on preferences over lotteries, utilities represent choices under uncertainty.
  \[
    U'(o)=\alpha U(o)+\beta,\qquad \alpha>0.
  \]
  Then
  \[
    EU'(a)=\alpha EU(a)+\beta.
  \]
  A positive scaling and a common shift preserve which action has the greatest expected utility.
  \equationlegend{$o$: outcome; $a$: action; $U,U'$: original/transformed utilities; $EU,EU'$: corresponding expected utilities. $\alpha$: positive scale; $\beta$: common offset. Outcome probabilities sum to $1$. A lottery is a probability distribution over outcomes.}
  \begin{keyidea}
    Arbitrary increasing transformations preserve rankings of certain outcomes, but can change decisions under uncertainty. Utility gaps matter.
  \end{keyidea}
\end{frame}

\begin{frame}[t]{Backup: One Engineering Approximation: Multiattribute Utility}
  \small
  Combine attribute scores on stated, comparable scales:
  \[
    U(o)=w_tu_t(o)+w_eu_e(o)+w_su_s(o)+w_pu_p(o).
  \]
  \equationlegend{$o$: outcome; $u_t,u_e,u_s,u_p$: scores for time, energy, safety, and progress, all oriented so higher is better. $w_t,w_e,w_s,w_p$: nonnegative tradeoff weights; here subscripts identify attributes, not time steps.}
  A larger weight matters relative to the score's scale: changing a score's units requires adjusting its weight.
  \begin{keyidea}
    An additive model assumes tradeoffs can be separated this way. Attribute interactions may require a richer utility model.
  \end{keyidea}
\end{frame}

\begin{frame}[t]{Backup: Value the Plan: Scan, Observe, Then Choose}
  \small
  Compare the best route now with the best route after each possible reading:
  \[
    V_{\text{now}}=\max_a EU(a\mid e),\qquad
    V_{\text{with}}=\sum_z P(z\mid e)\max_a EU(a\mid e,z).
  \]
  The \textbf{gross value of information} is $V_{\text{with}}-V_{\text{now}}$.
  Acquire the reading when this improvement exceeds its acquisition cost.
  \equationlegend{$e$: current evidence; $a$: route choice; $z$: possible reading; $EU$: expected utility. $V_{\text{now}}$: optimal value before sensing; $V_{\text{with}}$: expected optimal value after sensing, before its cost. Sum over readings; choose the route after each reading.}
  \begin{keyidea}
    A scan's immediate cost buys a later decision. Evaluating only its immediate reward would miss its benefit.
  \end{keyidea}
\end{frame}

\begin{frame}[t]{Backup: Will a Better Measurement Change the Decision?}
  \small
  The prior probability of blockage is $P(\text{Blocked})=0.35$.
  Without scanning, choose the detour: $V_{\text{now}}=45$.

  A scan costs $c=5$ utility units and has the sensor model
  \[
    P(+\mid\text{Blocked})=0.90,\qquad P(+\mid\text{Clear})=0.10.
  \]
  \equationlegend{$+$: sensor reports blockage; $-$: sensor reports clear. These are readings, not true states. $c$: fixed additive utility cost; $V_{\text{now}}$: value of the best route without another reading.}
  Assume the world stays fixed during this brief scan, both routes remain available, and its delay and energy cost are captured by $c$.
  \begin{keyidea}
    We are valuing one imperfect measurement before one route choice, using the same utilities as before.
  \end{keyidea}
\end{frame}

\begin{frame}[t]{Backup: The Scan Produces Two Possible Belief Updates}
  \small
  Weight each sensor likelihood by its prior state probability:
  \[
    P(+)=0.90(0.35)+0.10(0.65)=0.38,\qquad P(-)=1-0.38=0.62.
  \]
  Bayes' rule gives
  \[
    P(\text{Blocked}\mid+)=\frac{0.90(0.35)}{0.38}\approx0.829,
  \]
  \[
    P(\text{Blocked}\mid-)=\frac{0.10(0.35)}{0.62}\approx0.0565.
  \]
  \equationlegend{$P(+),P(-)$: reading probabilities; $P(\text{Blocked}\mid+)$ and $P(\text{Blocked}\mid-)$: posterior blockage beliefs. The negative-reading likelihood under blockage is $1-0.90=0.10$; denominators normalize each posterior.}
  \begin{keyidea}
    Predict how often each reading occurs, then update the belief separately for each branch.
  \end{keyidea}
\end{frame}

\begin{frame}[t]{Backup: After the Scan, Choose a Route for Each Reading}
  \small
  For a \textbf{positive} reading, blockage probability is about $0.829$:
  \[
    EU(\text{Shortcut}\mid+)\approx(1-0.829)(80)+0.829(-60)\approx-36.1.
  \]
  Choose the \textbf{detour}, whose utility is $45$.

  \medskip
  For a \textbf{negative} reading, blockage probability is about $0.0565$:
  \[
    EU(\text{Shortcut}\mid-)\approx(1-0.0565)(80)+0.0565(-60)\approx72.1.
  \]
  Choose the \textbf{shortcut}, since $72.1>45$.
  \equationlegend{$EU(a\mid+)$ and $EU(a\mid-)$: route values after the reading, before sensing cost. $80,-60$: shortcut utilities; $45$: detour utility. Displayed posteriors are rounded; calculations use full precision.}
  \begin{keyidea}
    The conditional plan is: positive $\rightarrow$ detour; negative $\rightarrow$ shortcut.
  \end{keyidea}
\end{frame}

\begin{frame}[t]{Backup: Separate Plan Value, Information Value, and Cost}
  \small
  Average the value of the best route in each reading branch:
  \[
    V_{\text{with}}\approx\underbrace{0.38(45)}_{+\,:\ \text{detour}}+
      \underbrace{0.62(72.1)}_{-\,:\ \text{shortcut}}\approx61.8.
  \]
  Gross information value: $61.8-45=16.8$.

  Scan-plan value after cost: $61.8-5=56.8$.

  Net improvement over acting now: $56.8-45=11.8$.
  \equationlegend{$V_{\text{with}}$: expected route utility after sensing, before cost. $0.38,0.62$: reading probabilities; $45,72.1$: optimized branch values; $5$: scan cost. All values use the same utility scale.}
  \begin{keyidea}
    The noisy-scan plan has value $56.8$, a gain of $11.8$ over acting now. Its gross information value $16.8$ is below the perfect-information bound $22.75$.
  \end{keyidea}
\end{frame}
