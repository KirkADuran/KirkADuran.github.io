\section{From Myopic Decisions to Sequential Decisions}

\begin{frame}{Lecture 22 Gave Us a Way to Compare Consequences}
  We combined:
  \begin{itemize}
    \item beliefs about an uncertain world;
    \item utilities that express preferences over outcomes;
    \item expected utility to compare a route choice;
    \item value of information to decide whether a reading was worth buying.
  \end{itemize}
  Those examples ended after one route segment.
  \begin{keyidea}
    Today we model the next state, the decisions available there, and the consequences of continuing to act.
  \end{keyidea}
\end{frame}

\begin{frame}{Today: Two Models for Sequential Decisions}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{block}{Part 1: MDPs}
        We observe the current modeled state.

        Actions have uncertain outcomes. Plan for accumulated reward and choose an action for each state.
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Part 2: POMDPs}
        The current physical state may be hidden.

        Maintain a belief, plan for future observations, and choose an action for each belief.
      \end{block}
    \end{column}
  \end{columns}
  \begin{keyidea}
    Both models optimize a strategy over time. Partial observability changes what the agent knows when choosing.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{The Problem: A Locally Good Action Can Create a Bad Future}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{block}{Shortcut now}
        Immediate reward: $+6$.

        Fast if the plaza stays clear, but entering it may trap the robot behind dense pedestrian traffic.
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Detour now}
        Immediate reward: $+2$.

        Slower right now, but it moves the robot into a predictable corridor with many good future choices.
      \end{block}
    \end{column}
  \end{columns}

  \begin{keyidea}
    The quality of an action depends on both its immediate outcome and the future states it makes likely.
  \end{keyidea}
\end{frame}

\begin{frame}{A Fixed Route Cannot Specify Every Future Response}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.53\textwidth}
      A path commits to one sequence of moves. But after each action:
      \begin{itemize}
        \item pedestrians may block the intended route;
        \item wheel slip may put the robot elsewhere;
        \item a door may open or close.
      \end{itemize}
      The robot must choose again from the situation it actually reaches.
    \end{column}
    \begin{column}{0.42\textwidth}
      \lectureimage[1.8in]{slide4.png}
    \end{column}
  \end{columns}
  \begin{keyidea}
    A policy specifies how to respond to the states reached, including outcomes that differ from the expected route.
  \end{keyidea}
\end{frame}

\section{Markov Decision Processes}

\begin{frame}[shrink=10]{A Markov Decision Process}
  \[
    \boxed{MDP=(S,A,T,R,\gamma)}
  \]

  \begin{center}
    \renewcommand{\arraystretch}{1.3}
    \begin{tabular}{c|p{0.72\textwidth}}
      $S$ & states: what situation is the robot in? \\
      $A$ & actions: what can the robot do? \\
      $T$ & transition model: where might an action lead? \\
      $R$ & reward: what immediate feedback is produced? \\
      $\gamma$ & discount factor: how strongly do future rewards matter? \\
    \end{tabular}
  \end{center}

  \begin{keyidea}
    An MDP separates the world model from the decision objective, then connects them through a policy.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{What Counts as a State?}
  A useful robot state might include:
  \begin{itemize}
    \item location or campus region;
    \item battery band;
    \item remaining delivery task;
    \item whether a door is open;
    \item congestion level if it changes future transitions.
  \end{itemize}

  The Markov property is
  \[
    P(s_{t+1}\mid s_t,a_t,\text{history})
    =P(s_{t+1}\mid s_t,a_t).
  \]
  \equationlegend{$P$: probability; $s_t,s_{t+1}$: states at successive time steps; $a_t$: current action; $\mid$: given. History means earlier states and actions; the current state and action suffice to predict the next state.}

  \begin{keyidea}
    ``State'' is an engineering abstraction: include what matters for predicting the future, omit what does not.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{State Design Is Already a Tradeoff}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{block}{State too small}
        If state is only location, but battery changes which moves are safe, then two visits to the ``same'' state can have different futures.

        The model is not truly Markov.
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{State too rich}
        Location $\times$ battery $\times$ task $\times$ crowd density $\times$ weather $\times$ door status $\times \cdots$

        Prediction improves, but the number of states explodes.
      \end{block}
    \end{column}
  \end{columns}

  \begin{keyidea}
    Markov state design trades model fidelity against state-space size.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{Actions Are the Robot's Controllable Choices}
  The action set might include:
  \begin{itemize}
    \item move forward, turn, backtrack;
    \item wait for traffic to move;
    \item perform a slower scan or re-localize;
    \item deliver the package;
    \item dock or request assistance.
  \end{itemize}

  \begin{keyidea}
    The action set defines what control authority the agent actually has.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{The Transition Model Makes Actions Stochastic}
  \[
    T(s,a,s')=P(s_{t+1}=s'\mid s_t=s,a_t=a)
  \]
  \equationlegend{$s$: current state; $a$: action; $s'$: candidate next state (the prime means next, not a derivative); $T(s,a,s')$: its transition probability. $t$: time step; $P(\cdot\mid\cdot)$: conditional probability.}

  Example from the campus intersection:
  \begin{center}
    \renewcommand{\arraystretch}{1.3}
    \begin{tabular}{l|l}
      \textbf{Action} & \textbf{Possible next states} \\\hline
      Shortcut & $0.70$ NearGoal, $0.30$ CrowdedPlaza \\
      Detour & $0.95$ SafeCorridor, $0.05$ CrowdedPlaza \\
      Wait & $0.60$ SafeCorridor, $0.40$ Intersection \\
    \end{tabular}
  \end{center}

  \begin{keyidea}
    The robot chooses the action, not the outcome.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{Reward Is Immediate; Return Is Long-Term}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{block}{Reward $R_t$}
        Feedback for one transition.
        \begin{itemize}
          \item $+20$ deliver package
          \item $-1$ each time step
          \item $-8$ enter dense crowd
          \item $-100$ collision
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Return $G_t$}
        Accumulated reward produced by a sequence of future decisions.

        A move can be mediocre now but create a much better long-term future.
      \end{block}
    \end{column}
  \end{columns}

  \equationlegend{$t$: current time step; $R_t$: reward for one transition; $G_t$: accumulated future reward starting at time $t$. Return means accumulated reward, not a single immediate reward.}
  \begin{keyidea}
    MDPs optimize a trajectory of consequences, not the next reward in isolation.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{Discounted Return}
  \[
    G_t=R_{t+1}+\gamma R_{t+2}+\gamma^2R_{t+3}+\cdots
  \]
  \equationlegend{$G_t$: return from time $t$; $R_{t+1}$: next reward; $R_{t+2},R_{t+3}$: later rewards; $\gamma$: discount factor; powers of $\gamma$ reduce the weight of progressively later rewards; $\cdots$: continue the pattern.}

  \begin{itemize}
    \item $\gamma=0$: only the next reward matters.
    \item $0<\gamma<1$: future rewards matter but are progressively discounted.
    \item $\gamma\approx1$: long-run consequences matter strongly.
  \end{itemize}

  \begin{keyidea}
    The discount factor controls how strongly delayed consequences influence current actions.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{What Does the Discount Factor Mean for the Robot?}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.30\textwidth}
      \begin{block}{Time preference}
        Earlier task completion may be operationally more valuable.
      \end{block}
    \end{column}
    \begin{column}{0.30\textwidth}
      \begin{block}{Planning emphasis}
        A smaller $\gamma$ emphasizes nearer rewards. Transition probabilities represent uncertain outcomes.
      \end{block}
    \end{column}
    \begin{column}{0.30\textwidth}
      \begin{block}{Mathematical control}
        For continuing tasks, $\gamma<1$ keeps bounded discounted returns finite.
      \end{block}
    \end{column}
  \end{columns}

  \begin{keyidea}
    $\gamma$ is not ``the probability the future exists.'' It is a modeling parameter controlling the value of delayed reward.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{A Policy Is Not a Path}
  \[
    \pi(s)=a
  \]
  \equationlegend{$\pi$: policy (decision rule); $s$: observed current state; $a$: action selected by the policy. $\pi(s)=a$ means choose action $a$ in state $s$.}

  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{block}{Path}
        Intersection $\rightarrow$ Plaza $\rightarrow$ Library $\rightarrow$ Goal.

        One assumed state sequence.
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Policy}
        If crowd low $\rightarrow$ shortcut.

        If crowd high $\rightarrow$ detour.

        If battery low $\rightarrow$ dock.
      \end{block}
    \end{column}
  \end{columns}

  \begin{keyidea}
    A policy specifies what to do from every state the robot might actually reach.
  \end{keyidea}
\end{frame}

  \small
  \begin{center}

\section{Long-Term Value}

\begin{frame}[shrink=10]{Immediate Reward Can Choose the Wrong Action}
  Shortcut:
  \[
    R=+6
  \]
  Detour:
  \[
    R=+2
  \]
  \equationlegend{$R$: immediate reward for the chosen action in this toy comparison. The values $+6$ and $+2$ do not include the rewards from later decisions.}

  A one-step agent prefers the shortcut immediately.

  But if the shortcut often enters a terrible future state, the correct long-run decision can reverse.

  \begin{keyidea}
    Scoring only the immediate reward omits the future decisions available from the next state.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{Action Value Adds the Future}
  Given continuation values $V(s')$ for the next states:
  \[
    Q(s,a)=\sum_{s'}T(s,a,s')\left[R(s,a,s')+\gamma V(s')\right]
  \]
  \equationlegend{$Q(s,a)$: action value given continuation values $V$; $s,a,s'$: current state, action, next state; $T$: transition probability; $R$: immediate transition reward; $\gamma$: discount factor; $V(s')$: continuation value; $\sum_{s'}$: sum over next states.}

  \begin{itemize}
    \item $T$: how likely is each next state?
    \item $R$: what do we receive now?
    \item $V(s')$: how good is the future state we create?
  \end{itemize}

  \begin{keyidea}
    An action is good when it gives good reward now and tends to place the robot in valuable future states.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{Bellman Optimality: The Core Recursion}
  \[
    V^*(s)=\max_a\sum_{s'}T(s,a,s')
    \left[R(s,a,s')+\gamma V^*(s')\right]
  \]
  \equationlegend{$V^*(s)$: optimal expected return from state $s$; $*$: optimal; $a$: candidate action; $s'$: next state; $T$: transition probability; $R$: immediate reward; $\gamma$: discount; $\sum_{s'}$: sum over next states; $\max_a$: take the greatest action value.}

  Read it procedurally:
  \begin{enumerate}
    \item consider one candidate action;
    \item consider every next state it can produce;
    \item add immediate reward to discounted future value;
    \item average by transition probability;
    \item choose the action with the largest result.
  \end{enumerate}

  \begin{keyidea}
    The value of a state is the value of making the best first move and then behaving optimally afterward.
  \end{keyidea}
\end{frame}

\begin{frame}{Worked Comparison: Immediate Reward Plus Future Value}
  \small
  Use $\gamma=0.9$ and these current continuation-value estimates:
  \begin{center}
    \renewcommand{\arraystretch}{1.2}
    \begin{tabular}{l|r}
      \textbf{Next state} & \textbf{Estimated future value} \\\hline
      Near Goal & $20$ \\
      Safe Corridor & $17$ \\
      Crowded Plaza & $-25$
    \end{tabular}
  \end{center}
  \[
    Q(\text{Intersection},\text{Shortcut})=6+0.9[0.70(20)+0.30(-25)]=11.85
  \]
  \[
    Q(\text{Intersection},\text{Detour})=2+0.9[0.95(17)+0.05(-25)]=15.41
  \]
  \equationlegend{$Q(s,a)$: immediate reward plus discounted expected continuation value; $\gamma=0.9$: discount. $0.70,0.30,0.95,0.05$: transition probabilities; $20,17,-25$: values from the table; $6,2$: immediate rewards for shortcut and detour, independent of next state.}

  \begin{keyidea}
    With these continuation values, detour wins despite its smaller immediate reward.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{What Is a Value Function?}
  \[
    V^\pi(s)=\mathbb{E}_\pi\left[G_t\mid S_t=s\right]
  \]
  \equationlegend{$V^\pi(s)$: expected return when following policy $\pi$ from state $s$; $\mathbb{E}_\pi$: average over trajectories generated by that policy; $G_t$: discounted return; $S_t=s$: start in state $s$ at time $t$; $V^*$: optimal value.}

  $V^\pi(s)$ asks:
  \begin{quote}
    If I start in state $s$ and follow policy $\pi$ from now on, how much discounted reward should I expect?
  \end{quote}

  $V^*(s)$ asks the same question assuming optimal behavior from now on.

  \begin{keyidea}
    Values compress many possible future trajectories into one reusable number.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{Value Iteration: Repeated Bellman Backups}
  \[
    V_{k+1}(s)=\max_a\sum_{s'}T(s,a,s')
    \left[R(s,a,s')+\gamma V_k(s')\right]
  \]
  \equationlegend{$k$: update sweep, not a time step; $V_k,V_{k+1}$: old and new value estimates; $s,a,s'$: state, action, next state; $T$: transition probability; $R$: immediate reward; $\gamma$: discount; $\sum_{s'}$: sum over next states; $\max_a$: best action value.}

  \begin{enumerate}
    \item initialize values, often $V_0(s)=0$;
    \item perform a Bellman backup for each state;
    \item repeat until values stabilize sufficiently.
  \end{enumerate}

  \begin{keyidea}
    Value iteration converts a long-horizon planning problem into repeated local updates.
  \end{keyidea}
\end{frame}

\begin{frame}{Intuition: Reward Propagates Backward}
  \begin{center}
    \resizebox{0.96\linewidth}{!}{%
    \begin{tikzpicture}[node distance=0.6cm,
      every node/.style={draw,rounded corners,align=center,minimum width=2.7cm,minimum height=1cm,font=\small}]
      \node (i) {Intersection\\$V_0=0$};
      \node[right=of i] (c) {Corridor\\$V_0=0$};
      \node[right=of c] (n) {Near Goal\\$V_0=0$};
      \node[right=of n] (g) {Terminal Goal\\$V=0$};
      \draw[-{Stealth},thick] (i)--(c);
      \draw[-{Stealth},thick] (c)--(n);
      \draw[-{Stealth},thick] (n)--node[above,draw=none,minimum width=0cm,minimum height=0cm] {$+20$} (g);
    \end{tikzpicture}}
  \end{center}
  \equationlegend{$V_0$: initial value estimate before any update sweep; $V$: future return from a state; $+20$: reward on entering the goal. The terminal goal has $V=0$ because no rewards remain after arrival.}

  In this separate, deterministic chain, all other rewards are zero and delivery pays $+20$ on entering the terminal goal.
  \begin{itemize}
    \item First sweep: Near Goal learns the value of delivering.
    \item Later sweeps carry discounted value back through Corridor and Intersection.
  \end{itemize}
  \begin{keyidea}
    The terminal state has no future reward. Dynamic programming reuses downstream values rather than enumerating every complete route.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{Extract the Policy from the Values}
  \[
    \pi^*(s)=\arg\max_a\sum_{s'}T(s,a,s')
    \left[R(s,a,s')+\gamma V^*(s')\right]
  \]
  \equationlegend{$\pi^*(s)$: optimal action in state $s$; $V^*(s')$: optimal next-state value; $T$: transition probability; $R$: immediate reward; $\gamma$: discount. $\sum_{s'}$: sum over next states; $\arg\max_a$: return a maximizing action, not its numerical value.}

  Example policy:
  \begin{itemize}
    \item Intersection $\rightarrow$ Detour
    \item SafeCorridor $\rightarrow$ Forward
    \item CrowdedPlaza $\rightarrow$ Wait / Exit
    \item NearGoal $\rightarrow$ Deliver
  \end{itemize}

  \begin{keyidea}
    The value function scores states; the policy chooses the action that best exploits those values.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{Policy Iteration: Another Route to the Same Goal}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{block}{Policy evaluation}
        Hold a policy fixed and estimate how valuable every state is if the robot keeps following it.
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Policy improvement}
        At each state, check whether a different action would produce greater expected long-term value.
      \end{block}
    \end{column}
  \end{columns}

  Repeat until the policy no longer changes.

  \begin{keyidea}
    Value iteration updates values directly; policy iteration alternates evaluating a policy and improving it.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{What an MDP Quietly Assumes}
  Standard MDP planning assumes:
  \begin{itemize}
    \item the current modeled state is known;
    \item a transition model is available;
    \item the state/action space is manageable enough to represent values or policies.
  \end{itemize}

  \begin{keyidea}
    MDPs solve sequential uncertainty in outcomes---but standard MDPs still assume the current state is observable.
  \end{keyidea}
\end{frame}

\section{Partial Observability}

\begin{frame}{Part 2: What If the Current State Is Hidden?}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.53\textwidth}
      So far we have assumed the current modeled state is observed. In practice:
      \begin{itemize}
        \item localization can drift;
        \item two hallways may look alike;
        \item a camera may miss an occluded pedestrian;
        \item a door may be hidden around a corner.
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \lectureimage[1.8in]{slide26.png}
    \end{column}
  \end{columns}
  \begin{keyidea}
    Uncertain action outcomes and an uncertain current state are different issues. A POMDP models both.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{Full Observability Was Doing More Work Than It Looked Like}
  An MDP policy is
  \[
    \pi(s).
  \]

  But the robot may only know
  \[
    P(CorridorA)=0.55,
    \qquad
    P(CorridorB)=0.45.
  \]
  \equationlegend{$\pi(s)$: action prescribed for a known state $s$; $P(\text{CorridorA}),P(\text{CorridorB})$: current probabilities of being in those two locations. Neither location is known with certainty.}

  Which state should we plug into the MDP policy?

  \begin{keyidea}
    If the current state is uncertain, a policy over physical states cannot be applied directly.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{But Lecture 21 Already Gave Us the Missing Object}
  \[
    b_t(s)=P(S_t=s\mid o_{1:t},a_{1:t-1})
  \]

  The \textbf{belief state} is a probability distribution over possible physical states.

  Example:
  \[
    b_t=[0.55,\;0.35,\;0.10]
  \]

  for Corridor A, Corridor B, and Plaza.
  \equationlegend{$b_t(s)$: belief probability for state $s$ at time $t$; $S_t$: hidden state variable; $o_{1:t}$: observations through $t$; $a_{1:t-1}$: earlier actions; $P(\cdot\mid\cdot)$: probability given that history. Vector entries follow the listed state order.}

  \begin{keyidea}
    State estimation and sequential decision making meet at the belief state.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{Partially Observable Markov Decision Process}
  \[
    \boxed{POMDP=(S,A,T,R,\Omega,O,\gamma)}
  \]
  \equationlegend{$S$: states; $A$: actions; $T$: transition model; $R$: reward model; $\gamma$: discount; $\Omega$: observation set; $O(o\mid s',a)$: probability of reading $o$ after action $a$ reaches next state $s'$.}

  Relative to an MDP we add:
  \begin{itemize}
    \item $\Omega$: the possible observations;
    \item $O(o\mid s',a)$: the observation model.
  \end{itemize}

  \begin{keyidea}
    A POMDP is an MDP where the agent acts without directly observing the underlying state.
  \end{keyidea}
\end{frame}

\begin{frame}{The POMDP as a Dynamic Bayes Network}
  \begin{center}
    \begin{tikzpicture}[>=Stealth,
      chance/.style={draw,ellipse,fill=CalPolyMint,minimum width=1.8cm,minimum height=0.7cm},
      choice/.style={draw,rectangle,fill=CalPolyBeige,minimum width=1.8cm,minimum height=0.7cm},
      reward/.style={draw,diamond,aspect=2.2,minimum width=1.8cm},
      edge/.style={->,thick}]
      \node[chance] (s) at (0,0) {$S_t$};
      \node[choice] (a) at (3,0.9) {$A_t$};
      \node[chance] (sp) at (6,0) {$S_{t+1}$};
      \node[chance] (o) at (9,0.9) {$O_{t+1}$};
      \node[reward] (r) at (3,-0.9) {$R_{t+1}$};
      \draw[edge] (s)--(sp);
      \draw[edge] (a)--(sp);
      \draw[edge] (sp)--(o);
      \draw[edge] (a)--(o);
      \draw[edge] (s)--(r);
      \draw[edge] (a)--(r);
      \draw[edge] (sp)--(r);
    \end{tikzpicture}
  \end{center}
  \equationlegend{$S_t,S_{t+1}$: hidden states; $A_t$: action; $O_{t+1}$: observation variable; $R_{t+1}$: reward; $t$: time step. The function $O(o\mid s',a)$ denotes the observation likelihood.}

  \small
  States are hidden; sensor readings are observed. Reward scores transitions; in this model we do not use a separate reward observation as evidence.
  \begin{keyidea}
    Actions influence future states and sensor evidence. Planning must anticipate both, using the transition and observation models.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{The Belief State Becomes the Planning State}
  Physical state:
  \begin{quote}
    where the robot actually is, whether the route is truly blocked, whether a door is truly open.
  \end{quote}

  Belief state:
  \[
    b=[0.55,\;0.35,\;0.10].
  \]

  The controller now uses
  \[
    \pi(b)
  \]

  instead of $\pi(s)$.
  \equationlegend{$b$: belief vector over Corridor A, Corridor B, and Plaza, in that order; $\pi(b)$: action chosen from the belief; $\pi(s)$: action chosen from a known physical state $s$.}

  \begin{keyidea}
    The robot plans from a probability distribution rather than pretending one hidden state is certainly true.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{Belief Update: Predict, Then Correct}
  Prediction after action $a_t$:
  \[
    b^-_{t+1}(s')=\sum_s T(s,a_t,s')b_t(s).
  \]

  Correction after observation $o_{t+1}$:
  \[
    b_{t+1}(s')\propto O(o_{t+1}\mid s',a_t)b^-_{t+1}(s').
  \]
  \equationlegend{$b_t$: current belief; $b^-_{t+1}$: predicted belief before sensing; $b_{t+1}$: corrected belief; $s,s'$: current and next states; $a_t$: action; $o_{t+1}$: new reading; $T$: transition model; $O$: observation likelihood; $\sum_s$: sum over current states; $\propto$: proportional to; normalize over next states to sum to one.}

  \begin{keyidea}
    Use the transition model to predict, then the observation likelihood to correct. This is filtering inside a sequential decision model.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{The Elegant Trick: A POMDP Becomes an MDP over Beliefs}
  \begin{center}
    \begin{tikzpicture}[node distance=0.65cm,every node/.style={draw,rounded corners,align=center,minimum width=2.5cm,minimum height=1.0cm,font=\small}]
      \node (b) {Belief\\$b_t$};
      \node[right=of b] (a) {Action\\$a_t$};
      \node[right=of a] (o) {Observation\\$o_{t+1}$};
      \node[right=of o] (bp) {New belief\\$b_{t+1}$};
      \draw[-{Stealth},thick] (b)--(a);
      \draw[-{Stealth},thick] (a)--(o);
      \draw[-{Stealth},thick] (o)--(bp);
    \end{tikzpicture}
  \end{center}
  \equationlegend{$b_t$: current belief distribution; $a_t$: chosen action; $o_{t+1}$: new observation; $b_{t+1}$: updated belief after acting and observing; $t$: time step.}

  The full action/observation history can be compressed into the current belief.

  \begin{keyidea}
    The belief is a sufficient statistic for POMDP control---but that elegance creates a new computational problem.
  \end{keyidea}
\end{frame}

\section{Why POMDPs Blow Up}

\begin{frame}[shrink=10]{Problem \#1: The Belief Space Is Continuous}
  Suppose there are only two possible hidden states: $A$ and $B$.

  A belief is
  \[
    b=[P(A),\;1-P(A)].
  \]

  Every value
  \[
    0\le P(A)\le1
  \]

  defines a different belief state.
  \equationlegend{$b$: belief vector; $P(A)$: probability of hidden state A; $1-P(A)$: probability of hidden state B. Here A and B label states, not action sets; $\le$: less than or equal to.}

  \begin{keyidea}
    Two physical states already create infinitely many possible beliefs.
  \end{keyidea}
\end{frame}

\begin{frame}{With Three Hidden States, Beliefs Fill a Triangle}
  \small
  For three hidden states,
  \[
    b=[p_1,p_2,p_3],
    \qquad
    p_1+p_2+p_3=1.
  \]
  \equationlegend{$b$: belief vector; $p_1,p_2,p_3$: nonnegative probabilities of Corridor A, Corridor B, and Plaza; their sum is $1$. $n$: number of hidden states; $n-1$: independent probability coordinates.}

  \begin{center}
    \begin{tikzpicture}
      \coordinate (a) at (0,0);
      \coordinate (b) at (5,0);
      \coordinate (c) at (2.5,2.0);
      \draw[thick,CalPolyGreen] (a)--(b)--(c)--cycle;
      \fill[CalPolyGold] (2.25,0.8) circle (2.2pt);
      \node[below] at (a) {Corridor A};
      \node[below] at (b) {Corridor B};
      \node[above] at (c) {Plaza};
      \node[right] at (2.25,0.8) {$[0.35,0.25,0.40]$};
    \end{tikzpicture}
  \end{center}

  \begin{keyidea}
    For $n$ hidden states, beliefs live in a continuous $(n-1)$-dimensional probability simplex.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{A POMDP Policy Maps Beliefs to Actions}
  Different belief regions can prefer different actions:
  \begin{itemize}
    \item highly confident in Corridor A $\rightarrow$ turn left;
    \item location ambiguous but uncertainty reducible $\rightarrow$ scan;
    \item likely near the plaza with high congestion $\rightarrow$ detour.
  \end{itemize}

  \begin{keyidea}
    The best action can change because the belief changes---even when the most likely physical state stays the same.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{Problem \#2: Future Observations Create Branching Contingencies}
  POMDP planning must reason about plans like:
  \begin{quote}
    Choose action $a_1$. If observation $o_1$ arrives, do $a_2$. If observation $o_2$ arrives instead, do $a_3$.
  \end{quote}

  So the future branches on:
  \begin{itemize}
    \item actions;
    \item stochastic physical transitions;
    \item stochastic observations;
    \item resulting belief updates.
  \end{itemize}

  \begin{keyidea}
    POMDP planning is contingency planning over both uncertain world evolution and uncertain future information.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{How Fast Does That Branching Grow?}
  A rough action--observation branching count is
  \[
    (|A||\Omega|)^H.
  \]

  With four actions, three coarse observation categories, and horizon $H=6$:
  \[
    (4\cdot3)^6=12^6\approx2.99\times10^6.
  \]
  \equationlegend{$|A|$: number of actions; $|\Omega|$: number of observation categories; vertical bars mean set size; $H$: planning horizon in steps. $(|A||\Omega|)^H$ counts possible length-$H$ action--observation sequences; $\approx$: approximately.}

  And every branch carries a continuously valued belief distribution.

  \begin{keyidea}
    Horizon, action branching, observation branching, and continuous beliefs compound each other.
  \end{keyidea}
\end{frame}

\section{Back to Monte Carlo}

\begin{frame}[shrink=10]{So We End Up Back at a Familiar Question}
  \begin{center}
    \Large\textbf{Do we really need to enumerate every possible future?}
  \end{center}

  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{block}{Exact approach}
        Represent and evaluate every relevant branch of the future.
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Monte Carlo approach}
        Sample plausible futures, observe their returns, and estimate which actions look best.
      \end{block}
    \end{column}
  \end{columns}

  \begin{keyidea}
    The same tradeoff returns: give up exact enumeration to gain tractable approximate decision making.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{Monte Carlo Rollouts for Decision Making}
  A simulation-based online planner can:
  \begin{enumerate}
    \item sample a physical state from the current belief;
    \item try a candidate action;
    \item sample the next state and observation;
    \item continue with a rollout policy that uses only the simulated observations or beliefs;
    \item accumulate discounted reward;
    \item repeat many times and average the return.
  \end{enumerate}

  \begin{keyidea}
    Average sampled returns to estimate action value. The simulated controller must not use the hidden state as if it were observed.
  \end{keyidea}
\end{frame}

\begin{frame}[shrink=10]{The Pieces We Already Learned Can Be Reused}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.30\textwidth}
      \begin{block}{Particle belief}
        Represent the current belief with particles instead of an exact table.
      \end{block}
    \end{column}
    \begin{column}{0.30\textwidth}
      \begin{block}{Monte Carlo futures}
        Sample transitions and observations to generate possible trajectories.
      \end{block}
    \end{column}
    \begin{column}{0.30\textwidth}
      \begin{block}{Tree search}
        Spend more simulation on promising choices rather than exploring everything uniformly.
      \end{block}
    \end{column}
  \end{columns}

  \begin{keyidea}
    Approximate POMDP planning can combine sampled beliefs with Monte Carlo lookahead instead of constructing the entire belief-space solution.
  \end{keyidea}
\end{frame}

  \begin{center}

  \end{keyidea}
\end{frame}
