\begin{frame}{Today: Follow a Changing Reference Diagram}
  Lecture 20 gave us the rules. Today we combine them in longer traces.
  \begin{itemize}
    \item Follow aliases as names move between lists.
    \item Separate a function's mutations from its returned value.
    \item Trace two parameters that may refer to the same list.
    \item Find the first unintended mutation and place a copy before it.
    \item Trace repeated calls using a binding ledger and object labels.
  \end{itemize}
  \begin{activity}
    Predict first, draw references, then compare with the reveal. Explain the statement responsible for each change.
  \end{activity}
\end{frame}

\begin{frame}{One-Minute Toolkit}
  \begin{center}
    \renewcommand{\arraystretch}{1.3}
    \begin{tabular}{l|l}
      \textbf{Statement} & \textbf{What to record} \\\hline
      \texttt{b = a} & Bind \texttt{b} to the object currently named by \texttt{a}. \\
      \texttt{b = a.copy()} & Create a new list; bind \texttt{b} to it. \\
      \texttt{a[0] = 7} & Change the existing list's contents. \\
      \texttt{a = a + [7]} & Create a new list; move the \texttt{a} arrow. \\
      \texttt{return a} & Give the caller a reference to the current object.
    \end{tabular}
  \end{center}
  \begin{keyidea}
    Label the lists we track L1, L2, \ldots\ as they appear. Keep a \textbf{binding ledger} and a separate record of each list's \textbf{contents}.
  \end{keyidea}
  Our examples use flat lists of integers: copies have independent list state.
\end{frame}

\begin{frame}{Trace 2: Different Arguments, Different Objects}
  \begin{center}
    \renewcommand{\arraystretch}{1.4}

\begin{frame}{Quick Check: A Copy as the Second Argument}
  Start again with \texttt{shared = [2, 5]}.
  \begin{itemize}
    \item \texttt{shared.copy()} is evaluated before entering \texttt{transfer}.
    \item \texttt{source} refers to L1, the original list.
    \item \texttt{target} refers to L2, the new list containing \texttt{[2, 5]}.
    \item The removal changes L1 to \texttt{[5]}.
    \item The append changes L2 to \texttt{[2, 5, 12]}.
  \end{itemize}
  \begin{block}{Caller-visible results}
    \texttt{shared == [5]} and the returned integer is \texttt{2}.\\
    The caller has not saved a reference to the copied list.
  \end{block}
\end{frame}

\begin{frame}{Practice A: Reveal}
  \begin{center}
    \renewcommand{\arraystretch}{1.4}
    \begin{tabular}{l|l|l}
      \textbf{Name} & \textbf{Final value} & \textbf{Reason} \\\hline
      \texttt{items, watch} & \texttt{[10, 20, 20]} & same list; index 1 receives \texttt{10 + 10} \\
      \texttt{saved} & \texttt{10} & integer read before the removal \\
      \texttt{snapshot} & \texttt{[20, 15, 10]} & separate sliced list, then reversed

  \end{keyidea}
\end{frame}

\begin{frame}[fragile]{Practice B: Rebinding Before the Next Mutation}
  \begin{block}{Code}
\begin{verbatim}
def revise(a: list[int], b: list[int]) -> list[int]:
    a.append(b.pop())
    b = b + [0]
    b[0] = 99
    return b

data = [1, 2, 3]
result = revise(data, data)
\end{verbatim}
  \end{block}
  \begin{activity}
    Predict \texttt{data}, \texttt{result}, and \texttt{data is result}. Record the state after \texttt{b.pop()} and again after \texttt{a.append(...)}.
  \end{activity}
\end{frame}

\begin{frame}{Practice B: Reveal}
  \begin{enumerate}
    \item On entry, \texttt{a}, \texttt{b}, and \texttt{data} refer to L1 = \texttt{[1, 2, 3]}.
    \item \texttt{b.pop()} removes and returns \texttt{3}; L1 becomes \texttt{[1, 2]}.
    \item \texttt{a.append(3)} puts it back; L1 becomes \texttt{[1, 2, 3]}.
    \item \texttt{b = b + [0]} creates L2 = \texttt{[1, 2, 3, 0]} and rebinds \texttt{b}.
    \item \texttt{b[0] = 99} changes L2, which is returned.
  \end{enumerate}
  \begin{block}{Final result}
    \texttt{data == [1, 2, 3]}\qquad \texttt{result == [99, 2, 3, 0]}\\
    \texttt{data is result} is \texttt{False}.
  \end{block}
  L1 was mutated twice even though its final contents match its initial contents.
\end{frame}

  \end{activity}

\begin{frame}{Exit Ticket: Reveal}
  \begin{center}
    \renewcommand{\arraystretch}{1.4}
    \begin{tabular}{l|l|l}
      \textbf{Name} & \textbf{Refers to} & \textbf{Final value} \\\hline
      \texttt{original} & L1 & \texttt{[4, 8, 8]} \\
      \texttt{alias, copy} & L2 & \texttt{[0]} \\
      \texttt{answer} & \texttt{None} & \texttt{None}
    \end{tabular}
  \end{center}
  \begin{itemize}
    \item \texttt{copy.pop()} removes \texttt{8} from L2; \texttt{append} adds it to L1 and returns \texttt{None}.
    \item \texttt{alias = copy} moves one arrow from L1 to L2.
    \item \texttt{alias[0] = 0} changes L2, visible through \texttt{copy} too.
  \end{itemize}
\end{frame}

\begin{frame}{What a Complete Mutation Trace Shows}
  \begin{itemize}
    \item The current list object for each active name and parameter.
    \item Intermediate contents after \emph{each} mutation, including inside expressions.
    \item The returned value separately from any change to an argument.
    \item A new box whenever slicing, copying, or concatenation creates a list.
    \item A moved arrow whenever a name is rebound.
    \item The exact first mutation that violates a function's promise.
  \end{itemize}
  \begin{keyidea}
    Follow the current arrows, execute one step at a time, and place a protective copy before the first mutation that needs isolation.
  \end{keyidea}
\end{frame}
